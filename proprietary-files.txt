# IMS (VoLTE) -- see docs/bringup/volte-stato-esperimento.md
#
# Libraries that the stock ROM also kept in /system/lib64 and that we lack:
# some live only in /vendor (and the system namespace cannot load them from
# there), others AOSP has removed outright -- libhwbinder and libhidltransport
# were folded into libhidlbase in Android 11.
#
# Without these, libmtk_vt_wrapper.so does not load and ImsService dies with
# UnsatisfiedLinkError as soon as it touches the video call provider.
system/lib64/vendor.mediatek.hardware.videotelephony@1.0.so
system/lib64/libvcodec_cap.so
system/lib64/libvcodec_capenc.so

# libhidltransport and libhwbinder must NOT be taken from stock: Android 12
# still builds them (deprecated, folded into libhidlbase) and copying them here
# would define them twice --
#   error: overriding commands for target .../libhidltransport.so
# They are added to PRODUCT_PACKAGES in device.mk instead, so the system
# versions are used, consistent with the rest of the build.

# com.android.ims.ImsServiceBase, which AOSP removed after Android 10 and which
# ImsService.apk still requires. It is installed under a different name so as
# not to clash with the ims-common.jar we build ourselves, which is needed.
system/framework/ims-common.jar:system/framework/mtk-ims-compat.jar

# FM radio.
#
# The LineageOS FMRadio app and the stock one are the same app: the library
# registers its native methods for exactly com/android/fmradio/FmNative, so the
# two fit together without adaptation. The driver is already running
# (fmradio_drv.ko loaded, /dev/fm present).
#
# In the stock ROM the library lives in product/lib*, which for us does not
# exist as a separate partition: it is installed into system. The product
# partition is not mounted under /mnt/stock, but extract-files.sh reads it from
# the image with debugfs, which needs no privileges.
-product/lib/libfmjni.so:system/lib/libfmjni.so
-product/lib64/libfmjni.so:system/lib64/libfmjni.so

# THE VENDOR PARTITION IS NOT HERE, AND THAT IS DELIBERATE
#
# For a while the 1622 files of the stock vendor were listed below, one by one,
# so that the build could construct the image. It does not any more: the stock
# image ships whole and untouched,
#
#   BOARD_PREBUILT_VENDORIMAGE := vendor/doogee/s88pro/vendor.img
#
# which is what the charter asks -- a maintainer must not require a *modified*
# prebuilt vendor image, and a non-A/B device may rely on an OEM one as long
# as it asserts its version at flash time, which releasetools.py does.
#
# Constructing it is not a choice we postponed, it is one that cannot be made
# in this tree, and the reason is measured: a vendor built against the blobs'
# VNDK 29 needs BOARD_VNDK_VERSION := 29, and soong refuses that with
#   error: bionic/libc/Android.bp: "libc" depends on undefined module
#          "vendor_snapshot"
# A vendor snapshot is generated only from a complete Android 10 tree with the
# sources of the vendor modules. Those sources do not exist -- it is the same
# wall this whole project was born against, the one that led to rebuilding the
# kernel from the factory binary.
#
# What the blobs would have cost, measured and written down in case the
# question ever comes back: 85 files overlap between the blobs and what the
# build installs, which is 42 decisions and not 85, and 1537 of the 1622 need
# no decision at all. The old list survives in git -- `git show
# 9ef0088:proprietary-files.txt` -- with its 84 exclusions already made.
#
# Only the system files stay here: the MediaTek framework .jars, without which
# mtkfusionrild keeps looping on the rild-oem socket, and the libraries that
# go with them.
# ---------------------------------------------------------------------------
